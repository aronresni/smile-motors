import Link from "next/link";
import { ROUTES } from "@/lib/constants";
import type { InventoryListItem } from "@/lib/admin/inventory";
import { StatusBadge } from "@/components/ui/status-badge";

function StatusPill({ status }: { status: string }) {
  return <StatusBadge domain="inventory" status={status} />;
}

function formatDate(iso: string | null): string {
  if (!iso) return "—";
  const d = new Date(iso);
  if (Number.isNaN(d.getTime())) return "—";
  return new Intl.DateTimeFormat("es-DO", { dateStyle: "medium" }).format(d);
}

function SaleCell({ item }: { item: InventoryListItem }) {
  if (!item.saleId) return <span className="text-muted-foreground">—</span>;
  return (
    <Link href={`${ROUTES.adminVentas}/${item.saleId}`} className="text-primary hover:underline">
      {item.saleNumber ?? "Ver venta"}
    </Link>
  );
}

function OriginCell({ item }: { item: InventoryListItem }) {
  return item.stockMode === "on_demand" ? "Bajo pedido" : "Stock físico";
}

function InventoryRow({ item }: { item: InventoryListItem }) {
  return (
    <tr className="border-b last:border-0 border-border/70">
      <td className="px-3 py-2.5 align-top">
        <Link href={`${ROUTES.adminInventario}/${item.inventoryUnitId}`} className="font-mono text-xs font-medium hover:underline">
          {item.vin ?? "—"}
        </Link>
      </td>
      <td className="px-3 py-2.5 align-top">
        <Link href={`${ROUTES.adminProductos}/${item.productId}`} className="hover:underline">
          {item.productName}
        </Link>
        {item.productLegacyId && <p className="text-[11px] text-muted-foreground">{item.productLegacyId}</p>}
      </td>
      <td className="px-3 py-2.5 align-top text-text-secondary">{item.variantName ?? "—"}</td>
      <td className="px-3 py-2.5 align-top"><StatusPill status={item.status} /></td>
      <td className="px-3 py-2.5 align-top"><SaleCell item={item} /></td>
      <td className="px-3 py-2.5 align-top text-muted-foreground">{formatDate(item.createdAt)}</td>
      <td className="px-3 py-2.5 align-top text-muted-foreground"><OriginCell item={item} /></td>
      <td className="px-3 py-2.5 text-right align-top">
        <Link
          href={`${ROUTES.adminInventario}/${item.inventoryUnitId}`}
          className="inline-flex items-center rounded-md border px-2.5 py-1 text-xs font-medium border-border hover:bg-surface-elevated"
        >
          Ver
        </Link>
      </td>
    </tr>
  );
}

function DesktopTable({ items }: { items: InventoryListItem[] }) {
  return (
    <div className="hidden overflow-hidden rounded-xl border border-border lg:block">
      <div className="overflow-x-auto">
        <table className="w-full text-sm">
          <thead>
            <tr className="border-b text-left text-[11px] uppercase tracking-wide text-muted-foreground border-border bg-surface">
              <th className="px-3 py-2.5 font-medium">VIN</th>
              <th className="px-3 py-2.5 font-medium">Producto</th>
              <th className="px-3 py-2.5 font-medium">Variante</th>
              <th className="px-3 py-2.5 font-medium">Estado</th>
              <th className="px-3 py-2.5 font-medium">Venta asociada</th>
              <th className="px-3 py-2.5 font-medium">Fecha</th>
              <th className="px-3 py-2.5 font-medium">Origen</th>
              <th className="px-3 py-2.5" />
            </tr>
          </thead>
          <tbody>
            {items.map((item) => (
              <InventoryRow key={item.inventoryUnitId} item={item} />
            ))}
          </tbody>
        </table>
      </div>
    </div>
  );
}

function MobileCards({ items }: { items: InventoryListItem[] }) {
  return (
    <ul className="space-y-2.5 lg:hidden">
      {items.map((item) => (
        <li key={item.inventoryUnitId} className="rounded-xl border p-4 border-border bg-surface">
          <div className="flex items-start justify-between gap-3">
            <Link href={`${ROUTES.adminInventario}/${item.inventoryUnitId}`} className="font-mono text-sm font-semibold hover:underline">
              {item.vin ?? "—"}
            </Link>
            <StatusPill status={item.status} />
          </div>
          <p className="mt-1 text-sm text-text-secondary">{item.productName}</p>
          <p className="text-[11px] text-muted-foreground">{item.variantName ?? "Sin variante"} · <OriginCell item={item} /></p>
          <div className="mt-2 flex items-center justify-between text-xs">
            <SaleCell item={item} />
            <span className="text-muted-foreground">{formatDate(item.createdAt)}</span>
          </div>
          <div className="mt-3">
            <Link
              href={`${ROUTES.adminInventario}/${item.inventoryUnitId}`}
              className="inline-flex items-center rounded-md border px-2.5 py-1 text-xs font-medium border-border hover:bg-surface-elevated"
            >
              Ver detalle
            </Link>
          </div>
        </li>
      ))}
    </ul>
  );
}

export function InventoryListResults({ items }: { items: InventoryListItem[] }) {
  return (
    <div>
      <DesktopTable items={items} />
      <MobileCards items={items} />
    </div>
  );
}
