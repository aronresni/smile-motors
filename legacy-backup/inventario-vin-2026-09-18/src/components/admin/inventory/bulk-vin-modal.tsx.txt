"use client";

import { useMemo, useState } from "react";
import { useRouter } from "next/navigation";
import { Button } from "@/components/ui/button";
import { Modal } from "@/components/ui/modal";
import { TextAreaField } from "@/components/ui/form-fields";
import { inventoryActionErrorText } from "@/lib/admin/inventory-errors";
import type { ProductOption } from "@/app/(admin)/admin/inventario/actions";
import { bulkCreateInventoryUnits } from "@/app/(admin)/admin/inventario/actions";
import { ProductVariantFields } from "@/components/admin/inventory/product-variant-fields";

interface BulkResult {
  createdCount: number;
  duplicateCount: number;
  invalidCount: number;
  duplicate: { vin: string; reason: string }[];
  invalid: { input: string; reason: string }[];
}

/** Vista previa PURAMENTE local (recorte + duplicados dentro del propio
 * texto pegado) — la existencia contra la base solo se conoce con
 * certeza al guardar; el resultado final tras guardar es el autoritativo
 * y siempre lista cada VIN, nunca descarta uno en silencio. */
function parsePreview(raw: string) {
  const lines = raw.split("\n").map((l) => l.trim());
  const seen = new Set<string>();
  let valid = 0;
  let duplicate = 0;
  let invalid = 0;
  for (const line of lines) {
    if (line === "") continue; // líneas vacías del textarea no cuentan
    const norm = line.toUpperCase();
    if (norm === "") {
      invalid++;
    } else if (seen.has(norm)) {
      duplicate++;
    } else {
      seen.add(norm);
      valid++;
    }
  }
  return { valid, duplicate, invalid, total: valid + duplicate + invalid };
}

export function BulkVinModal({ products }: { products: ProductOption[] }) {
  const router = useRouter();
  const [open, setOpen] = useState(false);
  const [productId, setProductId] = useState("");
  const [variantId, setVariantId] = useState("");
  const [raw, setRaw] = useState("");
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [result, setResult] = useState<BulkResult | null>(null);

  const preview = useMemo(() => parsePreview(raw), [raw]);

  const close = () => {
    if (busy) return;
    setOpen(false);
    setProductId("");
    setVariantId("");
    setRaw("");
    setError(null);
    setResult(null);
  };

  const submit = async () => {
    if (busy) return;
    if (!productId) {
      setError("Selecciona un producto.");
      return;
    }
    const vins = raw.split("\n").map((l) => l.trim()).filter((l) => l !== "");
    if (vins.length === 0) {
      setError("Ingresa al menos un VIN, uno por línea.");
      return;
    }
    setBusy(true);
    setError(null);
    const res = await bulkCreateInventoryUnits({ productId, variantId: variantId || undefined, vins });
    if (!res.ok) {
      setError(inventoryActionErrorText(res.code));
      setBusy(false);
      return;
    }
    setBusy(false);
    setResult(res as unknown as BulkResult);
    router.refresh();
  };

  return (
    <>
      <Button variant="secondary" size="sm" onClick={() => setOpen(true)}>
        Agregar varios VIN
      </Button>
      <Modal
        open={open}
        onClose={close}
        title="Agregar varios VIN"
        size="lg"
        description={result ? undefined : "Un VIN por línea. Se recortan espacios y se normalizan a mayúsculas."}
        footer={
          result ? (
            <Button variant="primary" size="sm" onClick={close}>Cerrar</Button>
          ) : (
            <>
              <Button variant="secondary" size="sm" onClick={close} disabled={busy}>Cancelar</Button>
              <Button variant="primary" size="sm" onClick={() => void submit()} disabled={busy || preview.total === 0}>
                {busy ? "Guardando…" : `Guardar ${preview.valid} VIN`}
              </Button>
            </>
          )
        }
      >
        {result ? (
          <div className="space-y-3">
            <div className="grid grid-cols-3 gap-3 text-center">
              <div className="rounded-lg p-3 bg-success/10">
                <p className="text-2xl font-semibold text-success">{result.createdCount}</p>
                <p className="text-[11px] text-success/80">Creados</p>
              </div>
              <div className="rounded-lg p-3 bg-warning/10">
                <p className="text-2xl font-semibold text-warning">{result.duplicateCount}</p>
                <p className="text-[11px] text-warning/80">Duplicados</p>
              </div>
              <div className="rounded-lg p-3 bg-danger/10">
                <p className="text-2xl font-semibold text-danger">{result.invalidCount}</p>
                <p className="text-[11px] text-danger/80">Inválidos</p>
              </div>
            </div>
            {result.duplicate.length > 0 && (
              <div>
                <p className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">Duplicados</p>
                <ul className="mt-1 space-y-0.5 text-xs text-text-secondary">
                  {result.duplicate.map((d, i) => (
                    <li key={i} className="font-mono">
                      {d.vin} <span className="text-muted-foreground">({d.reason === "ALREADY_EXISTS" ? "ya existía" : "repetido en el lote"})</span>
                    </li>
                  ))}
                </ul>
              </div>
            )}
            {result.invalid.length > 0 && (
              <div>
                <p className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">Inválidos</p>
                <ul className="mt-1 space-y-0.5 text-xs text-text-secondary">
                  {result.invalid.map((d, i) => (
                    <li key={i}>&quot;{d.input}&quot; — vacío</li>
                  ))}
                </ul>
              </div>
            )}
          </div>
        ) : (
          <div className="space-y-3">
            <ProductVariantFields
              products={products}
              productId={productId}
              variantId={variantId}
              onProductChange={setProductId}
              onVariantChange={setVariantId}
            />
            <TextAreaField
              label="VINs (uno por línea)"
              required
              rows={8}
              value={raw}
              onChange={(e) => setRaw(e.target.value)}
              placeholder={"VIN001\nVIN002\nVIN003"}
              className="font-mono"
            />
            {preview.total > 0 && (
              <p className="text-xs text-muted-foreground">
                Vista previa: <strong className="text-success">{preview.valid} válidos</strong>
                {preview.duplicate > 0 && <> · <strong className="text-warning">{preview.duplicate} duplicado{preview.duplicate === 1 ? "" : "s"}</strong></>}
                {preview.invalid > 0 && <> · <strong className="text-danger">{preview.invalid} inválido{preview.invalid === 1 ? "" : "s"}</strong></>}
                <span className="block text-[11px] text-muted-foreground">
                  Los duplicados YA existentes en la base solo se confirman al guardar.
                </span>
              </p>
            )}
            {error && <p role="alert" className="text-xs text-danger">{error}</p>}
          </div>
        )}
      </Modal>
    </>
  );
}
