"use server";

import { revalidatePath } from "next/cache";
import { createClient } from "@/lib/supabase/server";
import { requireZone } from "@/lib/auth/session";
import { ROUTES } from "@/lib/constants";

/**
 * Acciones de administración de inventario/VIN. TODAS pasan por RPC
 * `SECURITY DEFINER` que vuelven a exigir `is_admin()` server-side — mismo
 * patrón que Financieras/Vendedores/Productos. `requireZone("admin")` es la
 * primera barrera; la RPC (incluida la reserva atómica `UPDATE ... WHERE
 * status = 'AVAILABLE'`) es la segunda y definitiva, también a prueba de
 * condiciones de carrera entre dos administradores.
 */
export interface RpcResult {
  ok: boolean;
  code?: string;
  [key: string]: unknown;
}

function revalidateInventory(opts?: { unitId?: string; productId?: string; saleId?: string }) {
  revalidatePath(ROUTES.adminInventario);
  if (opts?.unitId) revalidatePath(`${ROUTES.adminInventario}/${opts.unitId}`);
  if (opts?.productId) revalidatePath(`${ROUTES.adminProductos}/${opts.productId}`);
  if (opts?.saleId) revalidatePath(`${ROUTES.adminVentas}/${opts.saleId}`);
}

export interface CreateInventoryUnitInput {
  productId: string;
  variantId?: string;
  vin: string;
  note?: string;
}

export async function createInventoryUnit(input: CreateInventoryUnitInput): Promise<RpcResult> {
  await requireZone("admin");
  const supabase = await createClient();
  const { data, error } = await supabase.rpc("admin_create_inventory_unit", {
    p_product_id: input.productId,
    p_variant_id: input.variantId || undefined,
    p_vin: input.vin,
    p_note: input.note || undefined,
  });
  if (error || !data) return { ok: false, code: "UNEXPECTED" };
  revalidateInventory({ productId: input.productId });
  return data as RpcResult;
}

export interface BulkCreateInventoryUnitsInput {
  productId: string;
  variantId?: string;
  vins: string[];
}

    p_product_id: input.productId,
    p_variant_id: input.variantId || undefined,
    p_vins: input.vins,
  });
  if (error || !data) return { ok: false, code: "UNEXPECTED" };
  revalidateInventory({ productId: input.productId });
  return data as RpcResult;
}

export async function updateInventoryVin(
  unitId: string,
  newVin: string,
  reason: string | undefined,
  productId?: string,
): Promise<RpcResult> {
  await requireZone("admin");
  const supabase = await createClient();
  const { data, error } = await supabase.rpc("admin_update_inventory_vin", {
    p_unit_id: unitId,
    p_new_vin: newVin,
    p_reason: reason || undefined,
  });
  if (error || !data) return { ok: false, code: "UNEXPECTED" };
  revalidateInventory({ unitId, productId });
  return data as RpcResult;
}

export async function assignInventoryVin(
  saleUnitId: string,
  inventoryUnitId: string,
  saleId: string,
): Promise<RpcResult> {
  await requireZone("admin");
  const supabase = await createClient();
  const { data, error } = await supabase.rpc("admin_assign_inventory_vin", {
    p_sale_unit_id: saleUnitId,
    p_inventory_unit_id: inventoryUnitId,
  });
  if (error || !data) return { ok: false, code: "UNEXPECTED" };
  revalidateInventory({ unitId: inventoryUnitId, saleId });
  return data as RpcResult;
}

export async function releaseInventoryVin(saleUnitId: string, reason: string, saleId: string): Promise<RpcResult> {
  await requireZone("admin");
  const supabase = await createClient();
  const { data, error } = await supabase.rpc("admin_release_inventory_vin", {
    p_sale_unit_id: saleUnitId,
    p_reason: reason,
  });
  if (error || !data) return { ok: false, code: "UNEXPECTED" };
  revalidateInventory({ saleId });
  return data as RpcResult;
}

export interface ProductOption {
  id: string;
  name: string;
  legacyId: string | null;
}

/** Catálogo completo (activo + inactivo) para los selectores de Admin de
 * este módulo — Admin puede dar de alta inventario para cualquier
 * producto, no solo los activos. */
export async function listAllProductsForPicker(): Promise<ProductOption[]> {
  await requireZone("admin");
  const supabase = await createClient();
  const { data, error } = await supabase
    .from("products")
    .select("id, name, legacy_id")
    .order("name");
  if (error || !data) return [];
  return data.map((p) => ({ id: p.id, name: p.name, legacyId: p.legacy_id }));
}

export interface VariantOption {
  id: string;
  colorName: string;
}

export async function listProductVariantsForPicker(productId: string): Promise<VariantOption[]> {
  await requireZone("admin");
  const supabase = await createClient();
  const { data, error } = await supabase
    .from("product_variants")
    .select("id, color_name")
    .eq("product_id", productId)
    .order("color_name");
  if (error || !data) return [];
  return data.map((v) => ({ id: v.id, colorName: v.color_name }));
}

export interface AvailableInventoryOption {
  id: string;
  vin: string | null;
}

/** Unidades AVAILABLE que calzan con el producto/variante EXACTOS de la
 * unidad de venta — usado por el modal "ASIGNAR VIN" en Admin Sale Detail.
 * Filtro de conveniencia solamente: la RPC de asignación vuelve a validar
 * la compatibilidad server-side (nunca se confía solo en este filtro). */
export async function listAvailableInventoryForAssignment(
  productId: string,
  variantId: string | null,
): Promise<AvailableInventoryOption[]> {
  await requireZone("admin");
  const supabase = await createClient();
  // Espejo de la regla de `admin_assign_inventory_vin`: si la unidad de
  // venta NO fijó variante, cualquier variante del producto es candidata
  // (el server no restringe en ese caso); si sí la fijó, solo esa variante.
  let query = supabase
    .from("inventory_units")
    .select("id, vin")
    .eq("product_id", productId)
    .eq("status", "AVAILABLE")
    .order("vin");
  if (variantId) query = query.eq("variant_id", variantId);
  const { data, error } = await query;
  if (error || !data) return [];
  return data;
}
